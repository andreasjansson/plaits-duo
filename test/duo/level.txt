===
in duo mode, AUX (split layout) follows the LEVEL input at 1V/octave, from -0.5V to 5V
===
printf 'tap both\ntap right\n' > split.txt
for volts in -0.5 0 1 2 3 4 5; do echo "${volts}V: $(./aux_note split.txt $volts)"; done
---
-0.5V: {{ n_minus }}
0V: {{ n0 }}
1V: {{ n1 }}
2V: {{ n2 }}
3V: {{ n3 }}
4V: {{ n4 }}
5V: {{ n5 }}
---
where
* abs(n_minus - (n0 - 6)) < 0.05
* abs(n1 - (n0 + 12)) < 0.05
* abs(n2 - (n0 + 24)) < 0.05
* abs(n3 - (n0 + 36)) < 0.05
* abs(n4 - (n0 + 48)) < 0.05
* abs(n5 - (n0 + 60)) < 0.05

===
with LEVEL at 0V, the two voices are at the same pitch
===
printf 'wait 500ms\ntap both\ntap right\nknob frequency 0.3\nknob harmonics 0.5\nknob timbre 0\nknob morph 0.4\ncv v_oct 0\ncv level 0\nwait 500ms\nclear\nwait 500ms\n' | plaits-sim -o unison.wav
plaits-analyze pitch unison.wav main | sed -n 's/note: /main: /p'
plaits-analyze pitch unison.wav aux | sed -n 's/note: /aux: /p'
---
main: {{ main }}
aux: {{ aux }}
---
where
* abs(main - aux) < 0.05

===
a 3% gain error of the LEVEL input stage makes the octaves 3% too wide before calibration
===
printf 'tap both\ntap right\ngain_error level 1.03\n' > split.txt
echo "1V: $(./aux_note split.txt 1)"
echo "3V: $(./aux_note split.txt 3)"
---
1V: {{ n1 }}
3V: {{ n3 }}
---
where
* abs(n3 - n1 - 24.72) < 0.05

===
holding the two buttons in duo mode calibrates LEVEL, with LED 7: patch 1V and press the left button, then 3V
===
plaits-sim <<'SCRIPT'
wait 500ms
tap both
wait 300ms
gain_error level 1.03
press both
wait 2.1s
release both
echo waiting for 1V:
leds 500ms
cv level 1
wait 300ms
tap left
echo waiting for 3V:
leds 500ms
cv level 3
wait 300ms
tap left
echo done:
leds
state
SCRIPT
---
waiting for 1V:
leds .......G
waiting for 3V:
leds .......Y
done:
leds G.......
duo_mode 1
engine 0
duo_engine 0
saved duo_mode 1 engine 0 duo_engine 0
level calibration scale -58.250 offset 47.571
color_blind 0

===
after calibration, the octaves are exact, and the calibration is kept when the module is powered off
===
cat > calibrate.txt <<'SCRIPT'
tap both
wait 300ms
gain_error level 1.03
press both
wait 2.1s
release both
wait 500ms
cv level 1
wait 300ms
tap left
wait 500ms
cv level 3
wait 300ms
tap left
reboot
wait 500ms
tap right
SCRIPT
echo "1V: $(./aux_note calibrate.txt 1)"
echo "3V: $(./aux_note calibrate.txt 3)"
---
1V: {{ n1 }}
3V: {{ n3 }}
---
where
* abs(n3 - n1 - 24) < 0.05
