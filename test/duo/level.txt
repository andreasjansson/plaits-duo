===
in duo mode, AUX (split layout) follows the LEVEL input at 1V/octave, from -0.5V to 5V
===
printf 'tap both\ntap right\n' > split.txt
for volts in -0.5 0 1 2 3 4 5; do echo "${volts}V: $(./aux_note split.txt $volts)"; done
---
-0.5V: {{ n_minus }}
0V: {{ n0 }}
1V: {{ n1 }}
2V: {{ n2 }}
3V: {{ n3 }}
4V: {{ n4 }}
5V: {{ n5 }}
---
where
* abs(n_minus - (n0 - 6)) < 0.05
* abs(n1 - (n0 + 12)) < 0.05
* abs(n2 - (n0 + 24)) < 0.05
* abs(n3 - (n0 + 36)) < 0.05
* abs(n4 - (n0 + 48)) < 0.05
* abs(n5 - (n0 + 60)) < 0.05

===
with LEVEL at 0V, the two voices are at the same pitch
===
printf 'wait 500ms\ntap both\ntap right\nknob frequency 0.3\nknob harmonics 0.5\nknob timbre 0\nknob morph 0.4\ncv v_oct 0\ncv level 0\nwait 500ms\nclear\nwait 500ms\n' | plaits-sim -o unison.wav
plaits-analyze pitch unison.wav main | sed -n 's/note: /main: /p'
plaits-analyze pitch unison.wav aux | sed -n 's/note: /aux: /p'
---
main: {{ main }}
aux: {{ aux }}
---
where
* abs(main - aux) < 0.05

===
a 3% gain error of the LEVEL input stage makes the octaves 3% too wide before calibration
===
printf 'tap both\ntap right\ngain_error level 1.03\n' > split.txt
echo "1V: $(./aux_note split.txt 1)"
echo "3V: $(./aux_note split.txt 3)"
---
1V: {{ n1 }}
3V: {{ n3 }}
---
where
* abs(n3 - n1 - 24.72) < 0.05

===
until it is calibrated, LEVEL reads volts as V/OCT does: the gain error of the converter, which the two inputs share, is corrected by the factory calibration of V/OCT
===
# A converter reading 2.7% high (STM32F373's datasheet: 2.4 to 3.1%), and the
# factory calibration of V/OCT, with 1V and 3V (in the original firmware).
cat > module.txt <<'SCRIPT'
hw adc_gain 1.027
press both
wait 2.1s
release both
cv v_oct 1
wait 300ms
tap left
cv v_oct 3
wait 300ms
tap left
unpatch v_oct
tap both
tap right
SCRIPT
echo "1V: $(./aux_note module.txt 1)"
echo "3V: $(./aux_note module.txt 3)"
---
1V: {{ n1 }}
3V: {{ n3 }}
---
where
* abs(n3 - n1 - 24) < 0.05

===
while LEVEL is calibrated, OUT and AUX carry two voltages in turn (about 0.5V and 4.4V at 8V full scale), half of the time each
===
printf 'wait 500ms\ntap both\npress both\nwait 2.1s\nrelease both\nclear\nwait 1s\n' | plaits-sim -o calibration.wav
plaits-analyze levels calibration.wav
---
main: rms={{ rms }} peak={{ peak }}
aux: rms={{ aux_rms }} peak={{ aux_peak }}
correlation: 1.000
---
where
* aux_rms == rms
* aux_peak == peak
* abs(peak * 8 - 4.39) < 0.01
* abs(rms ^ 2 - ((2000 / 32768) ^ 2 + (18000 / 32768) ^ 2) / 2) < 0.002

===
after calibration, the octaves are exact, and the calibration is kept when the module is powered off
===
cat > calibrate.txt <<'SCRIPT'
tap both
wait 300ms
gain_error level 1.03
press both
wait 2.1s
release both
patch v_oct main
wait 500ms
tap left
unpatch v_oct
patch level main
wait 500ms
tap left
unpatch level
reboot
wait 500ms
tap right
SCRIPT
echo "1V: $(./aux_note calibrate.txt 1)"
echo "3V: $(./aux_note calibrate.txt 3)"
---
1V: {{ n1 }}
3V: {{ n3 }}
---
where
* abs(n3 - n1 - 24) < 0.05
