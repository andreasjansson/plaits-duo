===
harmonic: the fundamental is the same in both channels
===
printf 'wait 500ms\ntap both\n' > model.txt
./taps left 4 >> model.txt
printf 'knob harmonics 0\nknob timbre 0\nknob morph 1\ncv v_oct -1\nwait 500ms\nclear\nwait 2s\n' >> model.txt
plaits-sim -o fundamental.wav < model.txt
plaits-analyze image fundamental.wav | grep correlation
---
correlation: +1.00
bass correlation: +1.00
