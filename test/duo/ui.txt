===
the MODEL CV is added to the selected model, with the same volts per model as in the original firmware
===
# The LED of the active model is lit, the one of the selected model blinks
# (leds 512ms: over a whole blink). Above 2.5V, the original firmware goes on
# to the red models, duo mode stays on its last model.
cat > model_cv.txt <<'SCRIPT'
cv model 0.33
wait 300ms
leds 512ms
cv model 1
wait 300ms
leds 512ms
cv model 2.33
wait 300ms
leds 512ms
cv model 2.67
wait 300ms
leds 512ms
SCRIPT
echo original:
{ echo "wait 500ms"; cat model_cv.txt; } | plaits-sim-v1.1
echo duo:
{ echo "wait 500ms"; echo "tap both"; cat model_cv.txt; } | plaits-sim
echo duo, split:
{ echo "wait 500ms"; echo "tap both"; echo "tap right"; cat model_cv.txt; } | plaits-sim
---
original:
leds GG......
leds G..G....
leds G......G
leds R.......
duo:
leds YY......
leds Y..Y....
leds Y......Y
leds Y......Y
duo, split:
leds RR......
leds R..R....
leds R......R
leds R......R
