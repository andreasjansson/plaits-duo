===

%skip(needs numpy and scipy) if: ! ${PYTHON:-python3} -c "import numpy, scipy"
===

cd "$CCTR_TEST_PATH/../.."

import sys
sys.path.insert(0, "rig/host")
import scipy.io.wavfile as wavfile
import telemetry
sr, y = wavfile.read(sys.argv[1])
frames, rate = telemetry.decode(y[:, 1] / 32768.0, sr, return_rate=True)
f = frames[-1]
print(f"{len(frames)} frames (version {f.version}), {rate:.1f} samples per second, margin {min(g.margin for g in frames):.2f}")
print(f"duo mode {f.duo_mode}, {f.layout} layout, model {f.duo_engine & 7}")
print("patched:", ",".join(j for j in telemetry.PATCHED if f.is_patched(j)))
print(f"V/OCT {f.cv_mean(0):.4f} (sd {f.cv_std(0):.4f}), LEVEL {f.cv_mean(1):.4f} (sd {f.cv_std(1):.4f})")
print(f"voices: {f.note:.2f} and {f.level_note:.2f} semitones")

===
the commands of the rig on FM: the model, the knobs, the CV readings, the patched inputs and the debug instrument, as the module's telemetry reports them and its outputs show
%skip(needs numpy and scipy) if: ! ${PYTHON:-python3} -c "import numpy, scipy"
===
cd "$CCTR_TEST_PATH/../.."
"${PYTHON:-python3}" - <<'PYTHON'
import sys
sys.path.insert(0, "rig/host")
import numpy as np
import control, pitch, rigio
rig = rigio.VirtualRig(rigio.load_config(), sim="build/host/plaits-sim-rig")
ctl = control.Control(rig)
ctl.telemetry(0)
first = ctl.frame(check=False)
ctl.engine(6, "split")
ctl.knob("harmonics", 0.25)
ctl.patched(trigger=False, level=True)
ctl.cv("v_oct", 0.125)
f = ctl.frame()
print(f"{f.commands - first.commands} commands, {f.command_errors} errors")
print(f"{f.layout} layout, model {f.duo_engine & 7}, harmonics knob {f.pots[1] / 65535:.3f}")
print("patched:", ",".join(j for j in ("fm", "timbre", "morph", "trigger", "level") if f.is_patched(j)))
print(f"V/OCT reads {f.cv[1] / 32768:.4f}")
ctl.cv("v_oct", None)
ctl.instrument("sine", 1000, 8000, 4000)
cap = rig.capture(0.5)
rms = np.sqrt(np.mean(cap.y ** 2, 0)) * 32768
print(f"sine: {pitch.frequency(cap.out, cap.sr):.1f}Hz, {round(rms[0], -1):.0f} and {round(rms[1], -1):.0f} rms")
ctl.instrument("constant", 1000, -2000)
cap = rig.capture(0.2)
print(f"constant: {cap.out[-1] * 32768:.0f} and {cap.aux[-1] * 32768:.0f} (the output stage inverts)")
ctl.instrument("off")
f = ctl.frame()
print(f"instrument {f.instrument}, no telemetry until asked: {len(control.telemetry.decode(rig.capture(2.5).aux, rig.sr))} frames")
rig.oc.glitch = 1.0   # a 1V spike in each symbol
ctl.knob("harmonics", 0.75)
f = ctl.frame()
print(f"with glitches: harmonics knob {f.pots[1] / 65535:.3f}, {f.command_errors} errors")
rig.close()

===
the state of the updater, on AUX in the same code as the telemetry, decoded by the rig
%skip(needs numpy and scipy) if: ! ${PYTHON:-python3} -c "import numpy, scipy"
===
cd "$CCTR_TEST_PATH/../.."
"${PYTHON:-python3}" - <<'PYTHON'
import sys
sys.path.insert(0, "rig/host")
import numpy as np
import telemetry
# As UpdaterStatus and StatusSample in plaits/rig/updater/updater.cc.
words = [telemetry.MAGIC_UPDATER, (1 << 16) | 8, 1, 123, 15, 2, 16000]
words.append(telemetry._checksum(words))
bits = [0] * 32 + [(w >> (31 - i)) & 1 for w in words for i in range(32)]
x = np.concatenate([[1, 1, 1, -1, -1, -1] if b else [-1, -1, -1, 1, 1, 1] for b in bits])
x = np.concatenate([np.zeros(1000), x * 0.3, np.zeros(1000)])
frames = telemetry.decode(x, telemetry.MODULE_RATE, kind=None)
print(len(frames), frames[0].kind, frames[0].summary())
PYTHON
---
1 updater updater: receiving, 123 packets, 15 pages written, 2 errors, peak 16000
